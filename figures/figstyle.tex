% Shared style for the figures. Each figure is built as its own PDF from
% figures/standalone/<name>.tex (run ./build_figures.sh, or compile that file in TeXstudio).
\usepackage{tikz}
\usetikzlibrary{arrows.meta,decorations.pathreplacing,calc}
\usepackage{pgfplots}
\pgfplotsset{compat=1.16}
\usepgfplotslibrary{groupplots}
\colorlet{cA}{blue!60!black}
\colorlet{cB}{red!75!black}
\colorlet{cC}{orange!80!black}
\colorlet{cD}{green!45!black}
\pgfplotsset{crease/.style={axis x line*=bottom, axis y line*=left, tick align=outside,
  every tick label/.append style={font=\footnotesize}, label style={font=\small},
  title style={font=\small}, legend style={font=\footnotesize, draw=none, fill=none},
  legend cell align=left, every axis plot/.append style={line width=0.9pt}}}
\pgfplotsset{creasesurf/.style={view={68}{24}, xmin=-1, xmax=1, ymin=-1, ymax=1,
  xtick={-1,0,1}, ytick={-1,0,1}, xlabel={$x/W$}, ylabel={$y/W$}, tick align=outside,
  every tick label/.append style={font=\footnotesize}, label style={font=\small},
  title style={font=\small}, scaled z ticks=false,
  zticklabel style={/pgf/number format/fixed},
  colormap={crease}{color=(blue!60!black) color=(white) color=(orange!80!black)}},
  surfstyle/.style={surf, shader=faceted, draw=black!35, line width=0.1pt, mesh/ordering=y varies}}
